    📚 توثيق API للمطورين
    الطريقةالمسارالوظيفة
    POST/api/jobsإنشاء مهمة دمج
    POST/api/upload?job=&source=&seq=رفع أجزاء الملفات (≤4MB)
    POST/api/jobs/:id/processتنفيذ الدمج كاملًا
    GET/api/jobs/:idالحالة + التقدم
    GET/api/jobs/:id/videoبث الناتج (Range) + ?download=1
    DELETE/api/jobs/:idحذف فوري شامل
    GET/api/configالحدود العامة + version

    مثال: إنشاء مهمة من روابط مباشرة ودمجها:

    curl -X POST https://YOUR-DOMAIN/api/jobs \
  -H "Content-Type: application/json" \
  -d '{"videos":["https://www.w3schools.com/html/mov_bbb.mp4","https://test-videos.co.uk/vids/bigbuckbunny/mp4/h264/360/Big_Buck_Bunny_360_10s_1MB.mp4"]}'
# → {"jobId":"...","statusUrl":"/api/jobs/...","processUrl":"/api/jobs/.../process"}

curl -X POST https://YOUR-DOMAIN/api/jobs/JOBID/process
# → {"status":"done","videoUrl":"...","downloadUrl":"...","size":...,"duration":...,"resolution":"640x360"}

curl https://YOUR-DOMAIN/api/jobs/JOBID/video?download=1 -o merged.mp4
